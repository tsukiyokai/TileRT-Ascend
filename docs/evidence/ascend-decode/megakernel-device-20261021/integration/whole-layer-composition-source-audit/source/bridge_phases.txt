   54 __aicore__ inline void Gate(TPipe& pipe, __gm__ bfloat16_t* input,
   55     __gm__ bfloat16_t* weight, __gm__ float* output, GM_ADDR tiling,
   56     int64_t rows, int64_t blocks)
   57 {
   58     if ASCEND_IS_AIC {
   59         const int64_t core = GetBlockIdx();
   60         if (core >= kExperts / kGateTile) {
   61             return;
   62         }
   63         TCubeTiling cube;
   64         auto* words = reinterpret_cast<uint32_t*>(&cube);
   65         const auto* source = reinterpret_cast<__gm__ uint32_t*>(tiling);
   66         for (int64_t i = 0; i < sizeof(cube) / sizeof(uint32_t); ++i) {
   67             words[i] = source[i];
   68         }
   69         GateMatmul mm;
   70         mm.Init(&cube, &pipe);
   71         GlobalTensor<bfloat16_t> a, b;
   72         GlobalTensor<float> c;
   73         a.SetGlobalBuffer(input, rows * kHidden);
   74         b.SetGlobalBuffer(weight, kHidden * kExperts);
   75         c.SetGlobalBuffer(output, rows * kExperts);
   76         for (int64_t tile = core; tile < kExperts / kGateTile; tile += blocks) {
   77             const int64_t column = tile * kGateTile;
   78             mm.SetOrgShape(Capacity(rows), kExperts, kHidden);
   79             mm.SetSingleShape(rows, kGateTile, kHidden);
   80             mm.SetTensorA(a, false);
   81             mm.SetTensorB(b[column], false);
   82             mm.IterateAll(c[column]);
   83             mm.End();
   84         }
   85     }
   86 }

   88 __aicore__ inline void TopK(TPipe& pipe, GM_ADDR logits, GM_ADDR bias,
   89     GM_ADDR probabilities, GM_ADDR ids, GM_ADDR system, int64_t rows)
   90 {
   91     if ASCEND_IS_AIV {

   97             router.Init(logits, bias, probabilities, ids, logits, system,

  100         }
  101     }

  118 __aicore__ inline void Epilogue(TPipe& pipe, GM_ADDR routed, GM_ADDR shared,
  119     GM_ADDR scaled, GM_ADDR combined, int64_t rows, int64_t blocks)
  120 {
  121     if ASCEND_IS_AIV {
  122         constexpr int64_t tile = 1024;
  123         TBuf<TPosition::VECCALC> buffer;
  124         pipe.InitBuffer(buffer, tile * 12);
  125         auto bytes = buffer.Get<uint8_t>();
  126         auto x = bytes.ReinterpretCast<bfloat16_t>();
  127         auto y = bytes[tile * 2].ReinterpretCast<bfloat16_t>();
  128         auto value = bytes[tile * 4].ReinterpretCast<float>();
  129         auto term = bytes[tile * 8].ReinterpretCast<float>();
  130         GlobalTensor<bfloat16_t> input, other, intermediate, output;
  131         input.SetGlobalBuffer((__gm__ bfloat16_t*)routed, rows * kHidden);
  132         other.SetGlobalBuffer((__gm__ bfloat16_t*)shared, rows * kHidden);
  133         intermediate.SetGlobalBuffer((__gm__ bfloat16_t*)scaled, rows * kHidden);
  134         output.SetGlobalBuffer((__gm__ bfloat16_t*)combined, rows * kHidden);
  135         for (int64_t index = GetBlockIdx(); index < rows * kHidden / tile;
  136              index += 2 * blocks) {
  137             const int64_t begin = index * tile;
  138             DataCopy(x, input[begin], tile);
  139             DataCopy(y, other[begin], tile);
  140             Fence<HardEvent::MTE2_V>();
  141             Cast(value, x, RoundMode::CAST_NONE, tile);
  142             Cast(term, y, RoundMode::CAST_NONE, tile);
  143             PipeBarrier<PIPE_V>();
  144             Muls(value, value, 2.5F, tile);
  145             PipeBarrier<PIPE_V>();
  146             Cast(x, value, RoundMode::CAST_RINT, tile);
  147             Fence<HardEvent::V_MTE3>();
  148             DataCopy(intermediate[begin], x, tile);
  149             PipeBarrier<PIPE_V>();
  150             Cast(value, x, RoundMode::CAST_NONE, tile);
  151             PipeBarrier<PIPE_V>();
  152             Add(value, value, term, tile);
  153             PipeBarrier<PIPE_V>();
  154             Cast(y, value, RoundMode::CAST_RINT, tile);
  155             Fence<HardEvent::V_MTE3>();
  156             DataCopy(output[begin], y, tile);
  157             Fence<HardEvent::MTE3_MTE2>();
  158         }
  159     }
