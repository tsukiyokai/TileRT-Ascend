   22 // Epoch, each sender's ready word, and payload occupy separate cache lines.
   23 // Both ready and payload have two planes; they share the same epoch parity.
   24 struct CommunicationLayout {
   25     uint64_t epoch;
   26     uint64_t ready;
   27     uint64_t payload;
   28     uint64_t rank_bytes;
   29     uint64_t plane_bytes;
   30     uint64_t bytes;
   31 };
   32 
   33 WHOLE_LAYER_COMMUNICATION_DEVICE constexpr CommunicationLayout
   34 MakeCommunicationLayout(int64_t capacity, int32_t ranks)
   35 {
   36     CommunicationLayout layout{};
   37     layout.epoch = 0;
   38     layout.ready = kCommunicationCacheLine;
   39     layout.payload = layout.ready + 2 * ranks * kCommunicationCacheLine;
   40     layout.rank_bytes = capacity * kCommunicationHidden * sizeof(uint16_t);
   41     layout.plane_bytes = ranks * layout.rank_bytes;
   42     layout.bytes = layout.payload + 2 * layout.plane_bytes;
   43     return layout;
   44 }
